// ==========================================
// كود Google Apps Script الشامل لمبادرة Future Makers
// ==========================================

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    "status": "success",
    "message": "Future Makers API Active & Ready!"
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // إنشاء العناوين تلقائياً إذا كان الشيت فارغاً
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "التاريخ والوقت", "اسم ولي الأمر", "رقم الهاتف", "رقم هاتف بديل", "البريد الإلكتروني", 
        "المحافظة", "المدينة/المركز", "اسم الطفل", "السن", "تاريخ الميلاد", 
        "المدرسة", "الصف الدراسي", "يمتلك Laptop", "يمتلك Gmail", 
        "خبرة برمجة سابقة", "تفاصيل الكورسات", "الاهتمامات", "الأهداف", 
        "ملاحظات صحية", "توجيهات إضافية", "مشروع الطفل", "المواعيد المفضلة", "الموافقة والإقرار"
      ]);
    }
    
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch(err) {
        data = e.parameter || {};
      }
    } else if (e && e.parameter) {
      data = e.parameter;
    }

    var timestamp = new Date().toLocaleString("ar-EG", { timeZone: "Africa/Cairo" });
    
    // إضافة بيانات التسجيل لشيت جوجل
    sheet.appendRow([
      timestamp,
      data.parent_name || "",
      data.parent_phone || "",
      data.parent_alt_phone || "",
      data.parent_email || "",
      data.governorate || "",
      data.city || "",
      data.child_name || "",
      data.child_age || "",
      data.child_dob || "",
      data.school || "",
      data.grade || "",
      data.has_laptop || "",
      data.has_gmail || "",
      data.prior_coding || "",
      data.course_details || "",
      data.interests || "",
      data.goals || "",
      data.health_notes || "",
      data.guidance_notes || "",
      data.child_project || "",
      data.preferred_schedule || "",
      data.terms_consent || ""
    ]);

    // إرسال إشعار فوري للبريد الإلكتروني المخصص
    var emailRecipient = "qs20261985@gmail.com";
    var subject = "🚀 تسجيل جديد في مبادرة Future Makers - " + (data.child_name || "طلب جديد");
    var htmlBody = `
      

        
تسجيل جديد في مبادرة Future Makers

        

        
اسم الطفل: ${data.child_name || "غير مدخل"} (${data.child_age || ""} سنة)

        
اسم ولي الأمر: ${data.parent_name || "غير مدخل"}

        
رقم الهاتف: ${data.parent_phone || "غير مدخل"}

        
المحافظة والمدينة: ${data.governorate || ""} - ${data.city || ""}

        
جهاز Laptop: ${data.has_laptop || ""}

        
المواعيد المفضلة: ${data.preferred_schedule || ""}

        
مشروع الطفل المتوقع: ${data.child_project || ""}

        

        فتح جدول البيانات
      

    `;

    MailApp.sendEmail({
      to: emailRecipient,
      subject: subject,
      htmlBody: htmlBody
    });

    return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
